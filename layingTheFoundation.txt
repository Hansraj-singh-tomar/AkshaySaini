// ------------------------- React Component -----------------------------

// Part - 2 (React Components)
// 1. Class base component
// 2. Functional components - If a function is returning a react element it becomes a functional component. 
 

// React Element - To change it into an functional component we have to use arrow function and use capital letter of function 
const reactElementHeading = (
    <h1 className='head' tabIndex="5">This is React Element</h1>
);

// Functional component 
const HeadingCmp = () => {
    return (
        <div className='container'>
            <h1 className='heading'>This is functional component</h1>
        </div>
    )
}

const root = ReactDOM.createRoot(document.getElementById("root"));

// we can't render functional component like this 
// root.render(HeadingCmp); // wrong way
root.render(<HeadingCmp/>) // right way // this is how we have to render functional component

// Generally we render react element like this 
root.render(reactElementHeading); 




// This is an Example of component composition
// -------------------- functional component inside functional component ------------------------
// Title functional component
const Title = () => (
    <h1 className='head' tabIndex={5}>
        Title component
    </h1>
);

// Functional component 
const HeadingCmp = () => {
    return (
        <div className='container'>
            // Component Composition - component inside another component
            <Title />
            {Title()} // This is also right way
            <Title></Title> // This is also right way
            <h1 className='heading'>This is functional component</h1>
        </div>
    )
}

const root = ReactDOM.createRoot(document.getElementById("root"));
root.render(<HeadingCmp />)




// --------------------- React element inside functional component ----------------
// To use React Element inside Function component in that case we will have to use curly brackets

const title = (
    <h1 className='head' tabIndex={5}>React element</h1>
)

const HeadingCmp = () => {
    return (
        <div className='container'>
            {title}
            <h1>React Element inside Functional components</h1>
        </div>
    )
}

const root = ReactDOM.createRoot(document.getElementById("root"));
root.render(<HeadingCmp />)



// ------------ React Element inside React Element ----------------------

const elem = <span>React Element</span>;

const title = (
    <div className='container'>
        {elem}
        <h1 className='head'>React Element inside React Element</h1>
    </div>
)
const root = ReactDOM.createRoot(document.getElementById("root"));
root.render(title)
